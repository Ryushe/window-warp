cmake_minimum_required(VERSION 3.25)
project(windowwarp_companion LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
set(WARP_KWIN_VERSION "" CACHE STRING "Exact compositor version used to build this package")
if(NOT WARP_KWIN_VERSION MATCHES "^[56]\\.[0-9]+\\.[0-9]+$")
    message(FATAL_ERROR "Set WARP_KWIN_VERSION to the exact supported KWin 5 or 6 version")
endif()
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
add_library(warp_kwin_sdk INTERFACE)
if(WARP_KWIN_VERSION VERSION_LESS 6)
    # KWin 5's development package omits its core plugin/input headers. Use the
    # matching distro source and its CMake-generated headers, never ABI copies.
    set(WARP_KWIN5_SOURCE_DIR "/opt/kwin-sdk/kwin-${WARP_KWIN_VERSION}/src" CACHE PATH "Exact KWin 5 source src directory")
    set(WARP_KWIN5_GENERATED_DIR "/opt/kwin-sdk/generated/src" CACHE PATH "Matching configured KWin 5 build src directory")
    foreach(header input.h plugin.h)
        if(NOT EXISTS "${WARP_KWIN5_SOURCE_DIR}/${header}")
            message(FATAL_ERROR "Missing exact KWin 5 source header: ${WARP_KWIN5_SOURCE_DIR}/${header}")
        endif()
    endforeach()
    foreach(header config-kwin.h kwin_export.h)
        if(NOT EXISTS "${WARP_KWIN5_GENERATED_DIR}/${header}")
            message(FATAL_ERROR "Configure the exact KWin 5 distro source first: missing ${WARP_KWIN5_GENERATED_DIR}/${header}")
        endif()
    endforeach()
    file(STRINGS "${WARP_KWIN5_GENERATED_DIR}/config-kwin.h" plugin_version REGEX "^#define KWIN_PLUGIN_VERSION_STRING ")
    if(NOT plugin_version STREQUAL "#define KWIN_PLUGIN_VERSION_STRING \"${WARP_KWIN_VERSION}\"")
        message(FATAL_ERROR "Generated KWin plugin ABI does not match WARP_KWIN_VERSION: ${plugin_version}")
    endif()
    find_package(Qt5 REQUIRED COMPONENTS Core DBus Widgets X11Extras)
    find_package(KF5Config REQUIRED)
    find_package(KF5WindowSystem REQUIRED)
    include(KDEInstallDirs5)
    find_library(WARP_KWIN_CORE NAMES libkwin.so.5 REQUIRED)
    target_include_directories(warp_kwin_sdk INTERFACE
        "${WARP_KWIN5_SOURCE_DIR}" "${WARP_KWIN5_SOURCE_DIR}/libkwineffects"
        "${WARP_KWIN5_GENERATED_DIR}" "${WARP_KWIN5_GENERATED_DIR}/libkwineffects")
    target_link_libraries(warp_kwin_sdk INTERFACE ${WARP_KWIN_CORE}
        Qt5::Core Qt5::DBus Qt5::Widgets Qt5::X11Extras KF5::ConfigCore KF5::ConfigGui KF5::WindowSystem)
else()
    find_package(KWin ${WARP_KWIN_VERSION} EXACT REQUIRED)
    find_package(Qt6 REQUIRED COMPONENTS Core DBus Widgets)
    include(KDEInstallDirs6)
    target_link_libraries(warp_kwin_sdk INTERFACE KWin::kwin Qt6::Core Qt6::DBus Qt6::Widgets)
endif()
configure_file(metadata.json.in metadata.json @ONLY)
add_library(windowwarp_companion MODULE companion.cpp)
target_link_libraries(windowwarp_companion PRIVATE warp_kwin_sdk)
target_include_directories(windowwarp_companion PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(windowwarp_companion PRIVATE WARP_KWIN_VERSION="${WARP_KWIN_VERSION}")
set_target_properties(windowwarp_companion PROPERTIES PREFIX "")
install(TARGETS windowwarp_companion DESTINATION "${KDE_INSTALL_PLUGINDIR}/kwin/plugins")
install(FILES README.md PROTOCOL.md COPYING DESTINATION "${KDE_INSTALL_DATADIR}/doc/window-warp/kwin-companion")
install(FILES companion.cpp compat.h CMakeLists.txt metadata.json.in README.md PROTOCOL.md COPYING
    DESTINATION "${KDE_INSTALL_DATADIR}/doc/window-warp/kwin-companion/source")

option(WARP_COMPANION_NATIVE_TESTS "Build the private-compositor test support plugin" OFF)
if(WARP_COMPANION_NATIVE_TESTS)
    add_library(windowwarp_companion_test MODULE tests/support.cpp)
    target_link_libraries(windowwarp_companion_test PRIVATE warp_kwin_sdk)
    target_include_directories(windowwarp_companion_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    set_target_properties(windowwarp_companion_test PROPERTIES PREFIX "")
endif()
